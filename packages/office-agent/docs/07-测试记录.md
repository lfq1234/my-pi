```powershell
cd C:/Users/LENOVO/Desktop/my-pi
C:/Users/LENOVO/.workbuddy/binaries/node/versions/22.22.2/node.exe --experimental-strip-types -e "import { readOfficeFile } from 'file:///C:/Users/LENOVO/Desktop/my-pi/packages/office-agent/src/core/document-reader.ts'; const files=['C:/Users/LENOVO/Desktop/test/中文季度总结.docx','C:/Users/LENOVO/Desktop/test/中文销售数据.xlsx','C:/Users/LENOVO/Desktop/test/中文业务汇报.pptx']; for (const f of files) { const result = await readOfficeFile(f); console.log(JSON.stringify({ file: result.document.fileName, kind: result.document.kind, summary: result.document.summary.slice(0,180) }, null, 2)); }"
```
